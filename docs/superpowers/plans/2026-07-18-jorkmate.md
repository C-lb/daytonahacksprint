# jorkmate Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Hinge-for-jobs PWA — swipe right on a real Workday posting and a Daytona-sandboxed agent applies with your prefilled profile; full Oxylabs → Kimi → Daytona → Nosana sponsor loop.

**Architecture:** Single Express server (`src/server.mjs`) serves a vanilla-JS PWA from `web/` and orchestrates sponsors via the existing `src/sponsors.mjs` clients. State = three JSON files in `data/` (seeded, committed). Apply progress streams over SSE.

**Tech Stack:** Node 18+ ESM, Express, `@daytona/sdk` (installed), vanilla JS/CSS PWA, existing `src/sponsors.mjs` (`reason`, `kimi`, `aiand`, `oxylabsGet`, `nosana`, `env`).

## Global Constraints

- Spec: `docs/superpowers/specs/2026-07-18-jorkmate-design.md` — read it first.
- Testing is **smoke-level by spec**: every task verifies via a run/curl step, not a unit-test suite.
- Demo never depends on live scraping/onboarding — seed data must always render a working app.
- ai& calls MUST send a browser `User-Agent` (Cloudflare 1010). `sponsors.mjs` handles LLM fallback; never call fetch to an LLM directly.
- Kimi = big `max_tokens` (≥4096). Quick/cheap LLM calls use `reason()` default (deepseek via ai&).
- Mobile UI: safe-area insets (`env(safe-area-inset-top/bottom)`) everywhere; anti-vibecode skill governs styling (invoke it in UI tasks); sentence-case copy, no em dashes, no ALL-CAPS eyebrows.
- Commit after every task; **push immediately** (live collaboration): `git pull --rebase --autostash && git push`.
- Never commit `.env` or real keys.

```js
// jorkmate server: serves the PWA + API, orchestrates the sponsor loop.
import express from 'express';
import { randomUUID } from 'node:crypto';
import { load, save } from './store.mjs';

const app = express();
app.use(express.json({ limit: '10mb' }));
app.use(express.static(new URL('../web', import.meta.url).pathname));

// --- SSE plumbing: per-application event buffer + live subscribers ---
const buffers = new Map();   // appId -> [{step, message, screenshot?}]
const subscribers = new Map(); // appId -> Set<res>
export function emitStep(appId, message, screenshot = null) {
  const ev = { t: Date.now(), message, screenshot };
  (buffers.get(appId) ?? buffers.set(appId, []).get(appId)).push(ev);
  const apps = load('applications');
  const a = apps.find((x) => x.id === appId);
  if (a) { a.steps.push({ t: ev.t, message }); if (screenshot) a.screenshot = screenshot; save('applications', apps); }
  for (const res of subscribers.get(appId) ?? []) res.write(`data: ${JSON.stringify(ev)}\n\n`);
}
export function setStatus(appId, status) {
  const apps = load('applications');
  const a = apps.find((x) => x.id === appId);
  if (a) { a.status = status; save('applications', apps); }
  emitStep(appId, `status:${status}`);
}

app.get('/api/profile', (_q, res) => res.json(load('profile')));
app.post('/api/profile', (req, res) => { save('profile', { ...load('profile'), ...req.body }); res.json(load('profile')); });

app.get('/api/deck', (_q, res) => {
  const p = load('profile');
  const applied = new Set(load('applications').map((a) => a.jobId));
  res.json(load('jobs').filter((j) => !applied.has(j.id) && (!p.industries.length || p.industries.includes(j.industry))));
});

app.get('/api/applications', (_q, res) => res.json(load('applications')));

app.post('/api/swipe', async (req, res) => {
  const { jobId, direction } = req.body;
  if (direction !== 'right') return res.json({ ok: true });
  const application = { id: randomUUID(), jobId, status: 'queued', steps: [], screenshot: null, createdAt: Date.now() };
  save('applications', [...load('applications'), application]);
  res.json({ ok: true, applicationId: application.id });
  runApply(application).catch((e) => { emitStep(application.id, `error: ${e.message}`); setStatus(application.id, 'failed'); });
});

// Stub until Task 6 wires the Daytona pipeline.
let runApply = async (application) => {
  setStatus(application.id, 'applying');
  emitStep(application.id, 'apply pipeline not wired yet (Task 6)');
  setStatus(application.id, 'failed');
};
export const setRunApply = (fn) => { runApply = fn; };

app.get('/api/apply/:id/stream', (req, res) => {
  res.writeHead(200, { 'Content-Type': 'text/event-stream', 'Cache-Control': 'no-cache', Connection: 'keep-alive' });
  for (const ev of buffers.get(req.params.id) ?? []) res.write(`data: ${JSON.stringify(ev)}\n\n`); // replay
  const set = subscribers.get(req.params.id) ?? subscribers.set(req.params.id, new Set()).get(req.params.id);
  set.add(res);
  req.on('close', () => set.delete(res));
});

const port = process.env.PORT || 3000;
app.listen(port, '0.0.0.0', () => console.log(`jorkmate on http://localhost:${port} (LAN: your-ip:${port})`));
```

```js
// Hinge-style matchmaking: score profile↔job on the Nosana-deployed model.
// Falls back to reason() (ai&) so the deck never blocks on Nosana.
import { chat, reason, env } from './sponsors.mjs';
import { load, save } from './store.mjs';

const PROMPT = (p, j) => `You are a job matchmaker. Given CANDIDATE and JOB, reply ONLY with JSON {"score": 0-100, "blurb": "<one flirty-but-professional sentence on why they match, no em dashes>"}.
CANDIDATE: ${JSON.stringify({ skills: p.skills, summary: p.summary, work_history: p.work_history })}
JOB: ${JSON.stringify({ title: j.title, company: j.company, description: j.description?.slice(0, 800) })}`;

const parse = (text) => {
  const m = text.match(/\{[\s\S]*\}/);
  const o = JSON.parse(m[0]);
  return { score: Math.max(0, Math.min(100, Number(o.score) || 0)), blurb: String(o.blurb || '').slice(0, 140) };
};

export async function scoreJob(profile, job) {
  const messages = [{ role: 'user', content: PROMPT(profile, job) }];
  try {
    if (!env.NOSANA_INFERENCE_URL) throw new Error('no nosana inference url');
    const m = await chat({ base: env.NOSANA_INFERENCE_URL, key: env.NOSANA_API_KEY, model: env.NOSANA_INFERENCE_MODEL || 'llama3', messages, max_tokens: 200, temperature: 0.4 });
    return { ...parse(m.content), via: 'nosana' };
  } catch {
    return { ...parse(await reason(messages, { max_tokens: 300 })), via: 'fallback' };
  }
}

export async function scoreAll() {
  const profile = load('profile');
  const jobs = load('jobs');
  for (const job of jobs.filter((j) => !j.match)) {
    try { job.match = await scoreJob(profile, job); save('jobs', jobs); console.log(`scored ${job.id}: ${job.match.score} (${job.match.via})`); }
    catch (e) { console.log(`score failed ${job.id}: ${e.message}`); }
  }
}
```

- [ ] **Step 1: Write `src/scrape.mjs`**

Workday tenants expose `POST <site>/wday/cxs/<tenant>/<site>/jobs` returning JSON — no HTML parsing. `oxylabsGet` is GET-only, so shell curl through the proxy with `-X POST`:

```js
// Scrape Workday postings through Oxylabs residential proxy.
// Workday CXS API: POST {site}/wday/cxs/{tenant}/{siteName}/jobs {"limit":20,"offset":0,"searchText":""}
import { execFile } from 'node:child_process';
import { env } from './sponsors.mjs';
import { load, save } from './store.mjs';

function oxylabsPost(url, body) {
  return new Promise((resolve, reject) => {
    execFile('curl', ['-s', '--max-time', '40', '-x', env.OXYLABS_PROXY, '-U', `${env.OXYLABS_USERNAME}:${env.OXYLABS_PASSWORD}`,
      '-X', 'POST', '-H', 'Content-Type: application/json', '-d', JSON.stringify(body), url],
      (err, stdout, stderr) => (err ? reject(new Error(stderr || err.message)) : resolve(stdout)));
  });
}

// https://nvidia.wd5.myworkdayjobs.com/SiteName -> cxs jobs endpoint
const cxsUrl = (siteUrl) => {
  const u = new URL(siteUrl);
  const tenant = u.hostname.split('.')[0];
  const site = u.pathname.split('/').filter(Boolean).pop();
  return `${u.origin}/wday/cxs/${tenant}/${site}/jobs`;
};

export async function scrapeAll() {
  const jobs = load('jobs');
  const seen = new Set(jobs.map((j) => j.url));
  let added = 0;
  for (const site of load('sites')) {
    try {
      const raw = await oxylabsPost(cxsUrl(site.url), { limit: 20, offset: 0, searchText: '', appliedFacets: {} });
      for (const p of JSON.parse(raw).jobPostings ?? []) {
        const url = new URL(site.url).origin + new URL(site.url).pathname + p.externalPath;
        if (seen.has(url)) continue;
        seen.add(url); added++;
        jobs.push({ id: `job-${Date.now()}-${added}`, company: site.company, title: p.title,
          location: p.locationsText ?? '', salary: null, url, industry: 'Software & Developer Tools',
          description: p.title, match: null });
      }
      console.log(`${site.company}: ok`);
    } catch (e) { console.log(`${site.company}: ${e.message}`); }
  }
  save('jobs', jobs);
  return added;
}

if (import.meta.url === `file://${process.argv[1]}`) scrapeAll().then((n) => console.log(`${n} new jobs`));
```

import json, sys, time

async function renderDeck() {
  const view = document.getElementById('view');
  const jobs = await api('/deck');
  if (!jobs.length) { view.innerHTML = '<p class="empty">No more jobs. Pull new ones from a live scrape.</p>'; return; }
  view.innerHTML = '<div class="stack" id="stack"></div>';
  const stack = document.getElementById('stack');
  jobs.slice(0, 3).reverse().forEach((job) => stack.append(card(job)));

  function card(job) {
    const el = document.createElement('article');
    el.className = 'card';
    el.innerHTML = `
      <div class="score">${job.match ? job.match.score : '–'}</div>
      <h2>${job.title}</h2>
      <p class="meta">${job.company} · ${job.location}</p>
      ${job.match ? `<p class="blurb">${job.match.blurb}</p>` : ''}
      <div class="stamp stamp-apply">Applied</div><div class="stamp stamp-pass">Pass</div>`;
    attachSwipe(el, job);
    return el;
  }

  function attachSwipe(el, job) {
    let x0 = 0, dx = 0;
    el.addEventListener('pointerdown', (e) => { x0 = e.clientX; el.setPointerCapture(e.pointerId); el.style.transition = 'none'; });
    el.addEventListener('pointermove', (e) => {
      if (!x0) return;
      dx = e.clientX - x0;
      el.style.transform = `translateX(${dx}px) rotate(${dx / 18}deg)`;
      el.querySelector('.stamp-apply').style.opacity = Math.max(0, dx / 90);
      el.querySelector('.stamp-pass').style.opacity = Math.max(0, -dx / 90);
    });
    el.addEventListener('pointerup', async () => {
      el.style.transition = 'transform .25s ease';
      if (Math.abs(dx) > 90) {
        const dir = dx > 0 ? 'right' : 'left';
        el.style.transform = `translateX(${dx > 0 ? 500 : -500}px) rotate(${dx / 10}deg)`;
        setTimeout(() => { el.remove(); renderDeck(); }, 220);
        const res = await api('/swipe', { method: 'POST', body: JSON.stringify({ jobId: job.id, direction: dir }) });
        if (dir === 'right') { const t = toast('agent applying…'); t.onclick = () => watchApply(res.applicationId); }
        else toast('passed');
      } else { el.style.transform = ''; el.querySelectorAll('.stamp').forEach((s) => (s.style.opacity = 0)); }
      x0 = 0; dx = 0;
    });
  }
}
